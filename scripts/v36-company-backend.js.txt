async function sha256HexServer(s){const b=await crypto.subtle.digest('SHA-256',new TextEncoder().encode(String(s||'')));return[...new Uint8Array(b)].map(x=>x.toString(16).padStart(2,'0')).join('')}
function publicCompanyView(v){if(!v)return v;const {writeTokenHash,...safe}=v;return safe}
async function canWritePublic(req,rec,bodyToken=''){if(!rec?.writeTokenHash)return false;const token=String(req.headers.get('x-kosif-write-token')||bodyToken||'').trim();if(token.length<24)return false;return(await sha256HexServer(token))===rec.writeTokenHash}
async function listPublicCompanies(env){
  if(!env?.DATA)return[];const l=await env.DATA.list({prefix:'kosif:company:public:',limit:1000});const out=[];
  for(const k of l.keys||[]){const v=await env.DATA.get(k.name,'json');if(v)out.push(publicCompanyView(v))}
  const legacy=(await env.DATA.get('kosif_companies_v1','json'))||[],shared=(await env.DATA.get('shared_state','json'))||{};for(const c of legacy){const id='legacy-'+String(c.id||c.name||'company').replace(/[^a-zA-Z0-9_-]+/g,'-');if(out.some(x=>x.id===id||x.name===c.name))continue;const baseState=shared?.state&&typeof shared.state==='object'?shared.state:shared;const stateCopy=JSON.parse(JSON.stringify(baseState||{}));stateCopy.entity={...(stateCopy.entity||{}),name:c.name||stateCopy.entity?.name||'شركة'};out.push({id,mode:'public',name:c.name||'شركة',createdAt:c.createdAt||new Date().toISOString(),updatedAt:c.updatedAt||c.createdAt||new Date().toISOString(),legacy:true,readOnly:true,company:{state:stateCopy}})}
  return out.sort((a,b)=>String(b.updatedAt||b.createdAt||'').localeCompare(String(a.updatedAt||a.createdAt||'')));
}
async function companiesApi(req,env,u){
  if(req.method==='GET'&&u.pathname==='/api/kosif/companies')return json({companies:await listPublicCompanies(env)});
  if(req.method==='POST'&&u.pathname==='/api/kosif/companies'){
    const b=await req.json();const mode=b.mode==='encrypted'?'encrypted':'public',now=new Date().toISOString();
    if(mode==='public'){
      const company=b.company&&typeof b.company==='object'?b.company:null,nm=String(company?.state?.entity?.name||company?.entity?.name||company?.name||'').trim(),token=String(b.writeToken||'').trim();if(!nm)return json({error:'اسم الشركة مطلوب'},400);if(token.length<24)return json({error:'Write token صالح مطلوب لإنشاء شركة عامة'},400);
      const storedList=await env.DATA.list({prefix:'kosif:company:public:',limit:1000});let existing=null;for(const k of storedList.keys||[]){const x=await env.DATA.get(k.name,'json');if(String(x?.name||'').trim()===nm){existing=x;break}}
      if(existing&&!await canWritePublic(req,existing,token))return json({error:'شركة عامة بنفس الاسم موجودة. افتحها للقراءة أو استخدم جهازًا يملك صلاحية الكتابة.'},409);
      const id=String(b.id||existing?.id||crypto.randomUUID()),rec={...(existing||{}),id,mode:'public',legacy:false,readOnly:false,name:nm,createdAt:existing?.createdAt||b.createdAt||now,updatedAt:now,writeTokenHash:existing?.writeTokenHash||await sha256HexServer(token),company};
      await env.DATA.put('kosif:company:public:'+id,JSON.stringify(rec));return json({ok:true,company:publicCompanyView(rec)},existing?200:201);
    }
    const lookup=String(b.lookup||'').toLowerCase();if(!/^[a-f0-9]{64}$/.test(lookup)||!b.ciphertext||!b.iv||!b.salt)return json({error:'بيانات التشفير غير مكتملة'},400);
    const rec={mode:'encrypted',lookup,ciphertext:String(b.ciphertext),iv:String(b.iv),salt:String(b.salt),createdAt:now,updatedAt:now,version:1};await env.DATA.put('kosif:company:private:'+lookup,JSON.stringify(rec));return json({ok:true,lookup},201);
  }
  const pub=u.pathname.match(/^\/api\/kosif\/companies\/([^/]+)$/);if(req.method==='PUT'&&pub){const id=decodeURIComponent(pub[1]),b=await req.json(),company=b.company&&typeof b.company==='object'?b.company:null;if(!company)return json({error:'بيانات الشركة مطلوبة'},400);const old=await env.DATA.get('kosif:company:public:'+id,'json');if(!old)return json({error:'الشركة غير موجودة أو سجل قديم للقراءة فقط'},404);if(!old.writeTokenHash)return json({error:'هذا السجل قديم ومتاح للقراءة فقط حتى تتم مطالبته بإجراء ترحيل صريح.'},403);if(!await canWritePublic(req,old,b.writeToken))return json({error:'لا تملك صلاحية تعديل هذه الشركة من هذا الجهاز.'},403);const rec={...old,id,mode:'public',name:company?.state?.entity?.name||company?.entity?.name||company?.name||old?.name||'شركة',createdAt:old.createdAt||new Date().toISOString(),updatedAt:new Date().toISOString(),company};await env.DATA.put('kosif:company:public:'+id,JSON.stringify(rec));return json({ok:true,company:publicCompanyView(rec)})}
  const m=u.pathname.match(/^\/api\/kosif\/company\/private\/([a-f0-9]{64})$/i);
  if(req.method==='GET'&&m){const val=await env.DATA.get('kosif:company:private:'+m[1].toLowerCase(),'json');return val?json(val):json({error:'الشركة غير موجودة أو الكود غير صحيح'},404)}
  if(req.method==='PUT'&&m){const b=await req.json(),lookup=m[1].toLowerCase();if(!b.ciphertext||!b.iv||!b.salt)return json({error:'بيانات التشفير غير مكتملة'},400);const old=await env.DATA.get('kosif:company:private:'+lookup,'json');const rec={mode:'encrypted',lookup,ciphertext:String(b.ciphertext),iv:String(b.iv),salt:String(b.salt),createdAt:old?.createdAt||new Date().toISOString(),updatedAt:new Date().toISOString(),version:1};await env.DATA.put('kosif:company:private:'+lookup,JSON.stringify(rec));return json({ok:true,lookup})}
  return null;
}
